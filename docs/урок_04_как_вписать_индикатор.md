# Урок 04: Как вписать индикатор в Backtrader

## 1. Краткое резюме

Урок разбирает механику подключения встроенных индикаторов (SMA, RSI, Stochastic) в стратегиях Backtrader на примере SBER (дневные свечи через moexalgo, 2021–2024). Показаны три варианта стратегии «перевёртыш»: чистая SMA, SMA + RSI + CrossOver, и SMA + RSI + CrossOver + Stochastic. Основной акцент — шаблон `next` с обработкой лонга, шорта и отсутствия позиции.

---

## 2. Ключевые концепции

- **Шаблон «перевёртыш»** — `next` с тремя ветками: `if not self.position`, `elif self.position.size > 0`, `elif self.position.size < 0`
- **`CrossOver(indicator, level)`** — возвращает `1` (пересечение снизу вверх) или `-1` (сверху вниз)
- **`self.data` vs `self.datas[0]`** — оба варианта работают при одном источнике данных
- **`self.dataclose[-1]`** — значение предыдущего бара (проверка что пересечение произошло именно на этом баре)
- **Параметры в `params`** — все периоды и уровни вынести в `params` для последующей оптимизации через Optuna

### Загрузка данных

```python
import backtrader as bt
import pandas as pd
from moexalgo import Ticker


def process_candles(data, ticker_name):
    df = pd.DataFrame(data)
    df.drop(columns=['end'], inplace=True)
    df.rename(columns={'begin': 'datetime'}, inplace=True)
    df['datetime'] = pd.to_datetime(df['datetime'])
    df['ticker'] = ticker_name
    df = df[['datetime', 'open', 'high', 'low', 'close', 'volume']]
    return df


ticker = 'SBER'
data = Ticker(ticker).candles(start='2021-02-23', end='2024-07-16', period='1d')
df = process_candles(data, ticker)
```

### Стратегия 1: SMA-перевёртыш (результат: 100 963.56)

```python
class TestStrategy(bt.Strategy):
    params = (
        ('maperiod', 37),
    )

    def log(self, txt, dt=None):
        dt = dt or self.datas[0].datetime.date(0)
        # print('%s, %s' % (dt.isoformat(), txt))

    def __init__(self):
        self.dataclose = self.datas[0].close
        self.order = None
        self.sma = bt.indicators.SimpleMovingAverage(self.datas[0], period=self.params.maperiod)

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [order.Completed]:
            if order.isbuy():
                self.log('Покупка выполнена, Цена: %.2f, На сумму: %.2f, Комиссия %.2f' %
                         (order.executed.price, order.executed.value, order.executed.comm))
                self.buyprice = order.executed.price
                self.buycomm = order.executed.comm
            else:
                self.log('Продажа выполнена, Цена: %.2f, На сумму: %.2f, Комиссия %.2f' %
                         (order.executed.price, order.executed.value, order.executed.comm))
            self.bar_executed = len(self)
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.log('Отмена ордера')
        self.order = None

    def notify_trade(self, trade):
        if not trade.isclosed:
            return
        self.log('Операционная прибыль, Общая %.2f, С учетом комиссии %.2f' %
                 (trade.pnl, trade.pnlcomm))

    def next(self):
        self.log('Close, %.2f' % self.dataclose[0])
        if self.order:
            return

        if not self.position:
            # Вход в лонг: цена пересекла SMA снизу вверх
            if self.dataclose[0] > self.sma[0] and self.dataclose[-1] <= self.sma[-1]:
                self.log('Ордер на покупку создан, %.2f' % self.dataclose[0])
                self.order = self.buy()
            # Вход в шорт: цена пересекла SMA сверху вниз
            elif self.dataclose[0] < self.sma[0] and self.dataclose[-1] >= self.sma[-1]:
                self.log('Ордер на продажу создан, %.2f' % self.dataclose[0])
                self.order = self.sell()

        elif self.position.size > 0:
            # Закрытие лонга: цена пересекла SMA сверху вниз
            if self.dataclose[0] < self.sma[0] and self.dataclose[-1] >= self.sma[-1]:
                self.log('Бай закрыт, %.2f' % self.dataclose[0])
                self.order = self.sell()

        elif self.position.size < 0:
            # Закрытие шорта: цена пересекла SMA снизу вверх
            if self.dataclose[0] > self.sma[0] and self.dataclose[-1] <= self.sma[-1]:
                self.log('Селл закрыт, %.2f' % self.dataclose[0])
                self.order = self.buy()


if __name__ == '__main__':
    cerebro = bt.Cerebro()
    cerebro.addstrategy(TestStrategy)

    data = df.copy()
    data["datetime"] = pd.to_datetime(data["datetime"])
    data.set_index("datetime", inplace=True)
    cerebro.adddata(bt.feeds.PandasData(dataname=data))

    cerebro.broker.setcash(100000.0)
    cerebro.addsizer(bt.sizers.FixedSize, stake=10)
    cerebro.broker.setcommission(commission=0.001)

    print('Стартовый портфель: %.2f' % cerebro.broker.getvalue())
    cerebro.run()
    print('Конечный портфель: %.2f' % cerebro.broker.getvalue())
    # Результат: 100 963.56
```

### Стратегия 2: SMA + RSI + CrossOver (результат: 100 000.00)

```python
class TestStrategy(bt.Strategy):
    params = (
        ('maperiod', 37),
        ('rsiperiod', 15),
        ('overbought', 75),
        ('oversold', 25),
    )

    def __init__(self):
        self.dataclose = self.datas[0].close
        self.order = None
        self.sma = bt.indicators.SimpleMovingAverage(self.datas[0], period=self.params.maperiod)
        self.rsi_ind = bt.indicators.RelativeStrengthIndex(self.data, period=self.params.rsiperiod)
        # CrossOver(индикатор, уровень): 1 — пересёк снизу, -1 — пересёк сверху
        self.cross_up_rsi = bt.indicators.CrossOver(self.rsi_ind, self.params.overbought)
        self.cross_down_rsi = bt.indicators.CrossOver(self.rsi_ind, self.params.oversold)

    # ... notify_order, notify_trade — аналогично стратегии 1

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.dataclose[0] > self.sma[0] and self.dataclose[-1] <= self.sma[-1] and self.cross_up_rsi == -1:
                self.order = self.buy()
            elif self.dataclose[0] < self.sma[0] and self.dataclose[-1] >= self.sma[-1] and self.cross_down_rsi == 1:
                self.order = self.sell()

        elif self.position.size > 0:
            if self.dataclose[0] < self.sma[0] and self.dataclose[-1] >= self.sma[-1] and self.rsi_ind[0] > 85:
                self.order = self.sell()

        elif self.position.size < 0:
            if self.dataclose[0] > self.sma[0] and self.dataclose[-1] <= self.sma[-1] and self.rsi_ind[0] < 15:
                self.order = self.buy()
```

### Стратегия 3: SMA + RSI + CrossOver + Stochastic (результат: 99 711.83)

```python
class TestStrategy(bt.Strategy):
    params = (
        ('maperiod', 37),
        ('rsiperiod', 15),
        ('overbought', 75),
        ('oversold', 25),
        ('percK', 14),
        ('percD', 3),
    )

    def __init__(self):
        self.dataclose = self.datas[0].close
        self.order = None
        self.sma = bt.indicators.SimpleMovingAverage(self.datas[0], period=self.params.maperiod)
        self.rsi_ind = bt.indicators.RelativeStrengthIndex(self.data, period=self.params.rsiperiod)
        self.cross_up_rsi = bt.indicators.CrossOver(self.rsi_ind, self.params.overbought)
        self.cross_down_rsi = bt.indicators.CrossOver(self.rsi_ind, self.params.oversold)
        # Stochastic: percK — быстрая линия, percD — сигнальная
        self.stoch = bt.indicators.Stochastic(self.data,
                                              period=self.params.percK,
                                              period_dfast=self.params.percD)

    def next(self):
        if self.order:
            return

        if not self.position:
            # Лонг: SMA пересечение + RSI пересёк overbought снизу + stoch.percK перепродан
            if (self.dataclose[0] > self.sma[0] and self.dataclose[-1] <= self.sma[-1]
                    and self.cross_up_rsi == 1 and self.stoch.percK[0] < 30):
                self.order = self.buy()
            # Шорт: SMA пересечение + cross_down_rsi == 0 + stoch.percK перекуплен
            elif (self.dataclose[0] < self.sma[0] and self.dataclose[-1] >= self.sma[-1]
                  and self.cross_down_rsi == 0 and self.stoch.percK[0] > 70):
                self.order = self.sell()

        elif self.position.size > 0:
            if self.dataclose[0] < self.sma[0] and self.dataclose[-1] >= self.sma[-1] and self.rsi_ind[0] > 85:
                self.order = self.sell()

        elif self.position.size < 0:
            if self.dataclose[0] > self.sma[0] and self.dataclose[-1] <= self.sma[-1] and self.rsi_ind[0] < 15:
                self.order = self.buy()
```

---

## 4. Важные замечания

- **`self.data` = `self.datas[0]`** — оба варианта работают при одном источнике данных
- **`CrossOver(rsi, level)`** — где `level` — число. Не путать с `CrossOver(line1, line2)` для кросса двух линий
- **`self.dataclose[-1]`** — проверка что именно на этом баре произошло пересечение (а не несколько баров назад)
- **Параметры RSI vs CrossOver** — `cross_up_rsi == -1` означает RSI пересёк уровень overbought СВЕРХУ вниз (выход из перекупленности). `cross_up_rsi == 1` — RSI пересёк overbought СНИЗУ вверх (вход в перекупленность)

## 5. Шпаргалка

| Действие | Код |
|----------|-----|
| Данные SBER daily | `Ticker('SBER').candles(start='2021-02-23', end='2024-07-16', period='1d')` |
| SMA | `bt.indicators.SimpleMovingAverage(self.datas[0], period=N)` |
| RSI | `bt.indicators.RelativeStrengthIndex(self.data, period=N)` |
| CrossOver с уровнем | `bt.indicators.CrossOver(self.rsi_ind, self.params.overbought)` |
| Stochastic | `bt.indicators.Stochastic(self.data, period=percK, period_dfast=percD)` |
| Линия percK | `self.stoch.percK[0]` |
| Проверка лонга | `elif self.position.size > 0:` |
| Проверка шорта | `elif self.position.size < 0:` |

## 6. Применение в боте (T-Invest API)

- **`CrossOver` как триггер** — избавляет от хранения массивов прошлых цен. Как только `self.cross_up_rsi[0] == 1` — бот формирует API-запрос к T-Invest на выставление ордера
- **Контроль направления позиции** — `self.position.size > 0 / < 0` не позволит открыть второй лонг поверх первого и упрощает реверсивные системы
- **Подтверждение через `[-1]`** — проверка `self.dataclose[-1] <= self.sma[-1]` обязательна: без неё стратегия будет генерировать сигналы на каждом баре, пока выполняется условие
