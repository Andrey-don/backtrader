# Урок 02: Источники данных для Backtrader

## 1. Краткое резюме

Урок разбирает подключение к разным источникам рыночных данных для Backtrader: moexalgo (MOEX), T-Invest API (Тинькофф), TradingView (price_loaders), Yahoo Finance (yfinance) и CCXT (криптобиржи). Каждый источник оформлен в виде готовой функции-загрузчика, которая возвращает DataFrame с колонками `datetime, open, high, low, close, volume` — стандартный формат для `bt.feeds.PandasData`.

---

## 2. Ключевые концепции

- **Единый формат** — все загрузчики возвращают DataFrame с одинаковой структурой: индекс `datetime`, колонки `open, high, low, close, volume`
- **`bt.feeds.PandasData`** — принимает любой DataFrame нужного формата независимо от источника
- **Timezone** — T-Invest возвращает tz-aware данные. Нужно `.dt.tz_localize(None)` перед передачей в Backtrader
- **`multi_level_index=False`** — обязательный параметр для `yf.download` в новых версиях yfinance

---

## 3. Код урока

### Moexalgo (MOEX — акции, фьючерсы)

```python
import pandas as pd
from moexalgo import Ticker


def process_candles(data, ticker_name):
    df = pd.DataFrame(data)
    df.drop(columns=['end'], inplace=True)
    df.rename(columns={'begin': 'datetime'}, inplace=True)
    df['datetime'] = pd.to_datetime(df['datetime'])
    df['ticker'] = ticker_name
    df = df[['datetime', 'open', 'high', 'low', 'close', 'volume']]
    return df


ticker = 'SBER'
data = Ticker(ticker).candles(start='2021-02-23', end='2024-07-16', period='1d')
df = process_candles(data, ticker)
print(df.head())
```

Параметр `period`: `'1d'` — дневные, `60` (число) — часовые (минуты), `'1h'` — часовые.

### T-Invest API (T-Bank / Тинькофф)

```python
from datetime import timedelta, datetime
import pandas as pd
from tinkoff.invest import CandleInterval, Client
from tinkoff.invest.utils import now

TOKEN = 'Ваш ключ'  # из .env через os.getenv("TINKOFF_TOKEN")

TIMEFRAMES = {
    '1m': CandleInterval.CANDLE_INTERVAL_1_MIN,
    '5m': CandleInterval.CANDLE_INTERVAL_5_MIN,
    '15m': CandleInterval.CANDLE_INTERVAL_15_MIN,
    '1h': CandleInterval.CANDLE_INTERVAL_HOUR,
    '1d': CandleInterval.CANDLE_INTERVAL_DAY,
    '1w': CandleInterval.CANDLE_INTERVAL_WEEK,
    '1M': CandleInterval.CANDLE_INTERVAL_MONTH,
}


def find_working_instrument(client, ticker, instrument_type="share"):
    instruments = client.instruments.find_instrument(query=ticker)
    for inst in instruments.instruments:
        if inst.instrument_type == instrument_type and inst.ticker == ticker:
            try:
                test = list(client.get_all_candles(
                    instrument_id=inst.uid,
                    from_=now() - timedelta(days=30),
                    interval=CandleInterval.CANDLE_INTERVAL_DAY,
                ))
                if len(test) > 0:
                    return inst.uid
            except:
                continue
    raise ValueError(f"Инструмент '{ticker}' не найден")


def get_tinvest_candles(ticker, instrument_type, timeframe, start_date=None, end_date=None):
    """
    instrument_type: "share" или "futures"
    timeframe: '1m', '5m', '15m', '1h', '1d', '1w', '1M'
    start_date / end_date: строки 'YYYY-MM-DD'
    """
    rows = []
    if isinstance(start_date, str):
        start_date = datetime.strptime(start_date, '%Y-%m-%d')
    if isinstance(end_date, str):
        end_date = datetime.strptime(end_date, '%Y-%m-%d')

    with Client(TOKEN) as client:
        uid = find_working_instrument(client, ticker, instrument_type)
        for candle in client.get_all_candles(
            instrument_id=uid,
            from_=start_date or now() - timedelta(days=30),
            to=end_date or now(),
            interval=TIMEFRAMES[timeframe],
        ):
            def to_float(q): return q.units + q.nano / 1e9
            rows.append([
                candle.time, to_float(candle.open), to_float(candle.high),
                to_float(candle.low), to_float(candle.close), candle.volume,
            ])

    df = pd.DataFrame(rows, columns=['datetime', 'open', 'high', 'low', 'close', 'volume'])
    df['datetime'] = df['datetime'].dt.tz_localize(None)  # убрать timezone
    return df


# Пример для акций:
df, uid = get_tinvest_candles("SBER", "share", "1h", "2024-01-01", "2024-12-31")

# Пример для фьючерсов:
# df, uid = get_tinvest_candles("BRK5", "futures", "1d", "2025-02-01", "2025-02-28")
```

### TradingView (через price_loaders)

```python
from price_loaders.tradingview import load_asset_price
import pandas as pd


def process_tradingview_data(symbol, bars, timeframe, exchange=None):
    df = load_asset_price(symbol, bars, timeframe, exchange)
    df['time'] = pd.to_datetime(df['time']).dt.tz_localize(None) - pd.Timedelta(hours=3)
    df.rename(columns={'time': 'datetime'}, inplace=True)
    df = df[['datetime', 'open', 'high', 'low', 'close', 'volume']]
    return df


# Фьючерс на золото, 2400 часовых баров
df = process_tradingview_data("GD1!", 2400, "1H")

# Крипта
# df = process_tradingview_data("BTCUSDT", 2400, "1H", "BINANCE")
```

`-pd.Timedelta(hours=3)` — приведение UTC к московскому времени (UTC+3).

### Yahoo Finance (yfinance)

```python
import yfinance as yf
import pandas as pd


def process_yf_candles(data):
    df = data.copy()
    if isinstance(df.columns, pd.MultiIndex):
        df.columns = [x.replace(' ', '') for x in df.columns.get_level_values(0)]
    else:
        df.columns = [x.replace(' ', '') for x in df.columns]
    df = df.reset_index()
    df.rename(columns={
        'Date': 'datetime', 'Open': 'open', 'High': 'high',
        'Low': 'low', 'Close': 'close', 'Volume': 'volume',
    }, inplace=True)
    return df[['datetime', 'open', 'high', 'low', 'close', 'volume']]


def get_yf_candles(symbol, start_date, end_date, interval='1d'):
    data = yf.download(symbol, start=start_date, end=end_date,
                       interval=interval, multi_level_index=False)
    if not data.empty:
        return process_yf_candles(data)
    return None


df = get_yf_candles("AAPL", "2022-01-01", "2022-12-31")
```

### CCXT (крипта через Binance и другие биржи)

```python
import ccxt
import pandas as pd
from datetime import datetime, timedelta


def process_crypto_candles(data, ticker_name):
    df = pd.DataFrame(data, columns=['datetime', 'open', 'high', 'low', 'close', 'volume'])
    df['datetime'] = pd.to_datetime(df['datetime'], unit='ms')
    df['ticker'] = ticker_name
    return df[['datetime', 'open', 'high', 'low', 'close', 'volume']]


def get_crypto_candles(ticker, start_date, end_date, timeframe='1d'):
    """
    ticker: торговая пара, например 'BTC/USDT'
    timeframe: '1m','5m','15m','30m','1h','4h','1d','1w','1M'
    """
    exchange = ccxt.binance()
    since = exchange.parse8601(start_date.isoformat())
    until = exchange.parse8601(end_date.isoformat())
    ohlcv = exchange.fetch_ohlcv(ticker, timeframe, since, until)
    return process_crypto_candles(ohlcv, ticker.split('/')[0])


start_date = datetime.now() - timedelta(days=100)
end_date = datetime.now()
df = get_crypto_candles('BTC/USDT', start_date, end_date, '1d')
```

## 4. Важные замечания

- **`multi_level_index=False`** — обязательно для `yf.download` в yfinance ≥ 0.2. Без него колонки возвращаются как мультиндекс
- **Timezone T-Invest** — свечи приходят с UTC timezone. `dt.tz_localize(None)` убирает tzinfo; `dt.tz_convert(None)` — если уже tz-aware
- **moexalgo `period`** — принимает как строку `'1d'`, `'1h'`, так и число минут: `60`, `15`, `5`
- **CCXT `fetch_ohlcv`** — возвращает timestamp в миллисекундах. Конвертация: `pd.to_datetime(ts, unit='ms')`

## 6. Применение в боте (T-Invest API)

- **Основной источник для MOEX** — T-Invest API (`get_tinvest_candles`). Тот же токен используется для live-торговли
- **Исторические данные** — для бэктеста берём данные за 6–24 месяца, передаём в Cerebro
- **Sandbox** — при `TINKOFF_SANDBOX=true` исторические данные работают так же, как в production
- **Резерв для крипты** — CCXT + Binance для бэктеста крипто-стратегий без доступа к TradingView
