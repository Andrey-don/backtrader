# Урок 16: Мультиоптимизатор Optuna для Backtrader

## 1. Краткое резюме

Урок посвящён оптимизации стратегии сразу по нескольким инструментам в цикле. Написана функция-обёртка, которая принимает DataFrame как аргумент (через `lambda`), и цикл по списку тикеров — для каждого запускается отдельный Optuna study. Результаты всех оптимизаций сохраняются в CSV-файл. Данные — через `price_loaders.tradingview` (крипта, фьючерсы, Forex, SBER).

---

## 2. Ключевые концепции

- **`objective(trial, df)`** — функция принимает DataFrame как второй аргумент. Передаётся через `lambda trial: objective(trial, df)`
- **Цикл по тикерам** — для каждого символа создаётся свой `study`, данные загружаются отдельно
- **`try / except`** — защита от ошибок при загрузке несуществующих тикеров («ШВАБРА»)
- **Сохранение в CSV** — `df_results.to_csv('optuna_results.csv')` — таблица с лучшими параметрами по каждому инструменту
- **`optuna.logging.set_verbosity(optuna.logging.WARNING)`** — отключает подробный лог Optuna

---

## 3. Код урока

```python
import backtrader as bt
import pandas as pd
import optuna
import warnings

optuna.logging.set_verbosity(optuna.logging.WARNING)
warnings.filterwarnings('ignore')

# from price_loaders.tradingview import load_asset_price


def get_tradingview_data(symbol: str, bars: int, timeframe: str, exchange: str = None) -> pd.DataFrame:
    """Загрузка данных с TradingView"""
    df = load_asset_price(symbol, bars, timeframe, exchange)
    if df is None or df.empty:
        raise ValueError(f"Не удалось загрузить данные для {symbol}.")
    df['time'] = pd.to_datetime(df['time']).dt.tz_localize(None) - pd.Timedelta(hours=3)
    df.rename(columns={'time': 'datetime'}, inplace=True)
    df = df[['datetime', 'open', 'high', 'low', 'close', 'volume']]
    return df


class MyStrategy(bt.Strategy):
    params = (
        ('short_ma', 10),
        ('long_ma', 20),
        ('take_profit', 0.02),
        ('stop_loss', 0.01),
    )

    def __init__(self):
        self.short_ma = bt.indicators.SimpleMovingAverage(self.data.close, period=self.params.short_ma)
        self.long_ma = bt.indicators.SimpleMovingAverage(self.data.close, period=self.params.long_ma)
        self.dataclose = self.datas[0].close
        self.order = None

    def next(self):
        if self.position:
            if self.position.size > 0:
                if self.data.close[0] >= self.position.price * (1 + self.params.take_profit):
                    self.close()
                elif self.data.close[0] <= self.position.price * (1 - self.params.stop_loss):
                    self.close()
            elif self.position.size < 0:
                if self.data.close[0] <= self.position.price * (1 - self.params.take_profit):
                    self.close()
                elif self.data.close[0] >= self.position.price * (1 + self.params.stop_loss):
                    self.close()

        elif not self.position and self.short_ma[0] > self.long_ma[0] and self.short_ma[-1] <= self.long_ma[-1]:
            self.buy()
        elif not self.position and self.short_ma[0] < self.long_ma[0] and self.short_ma[-1] >= self.long_ma[-1]:
            self.sell()


def objective(trial, df):
    """Целевая функция — принимает DataFrame как аргумент"""
    short_ma = trial.suggest_int('short_ma', 9, 25)
    long_ma = trial.suggest_int('long_ma', 33, 65)
    take_profit = trial.suggest_float('take_profit', 0.01, 0.05)
    stop_loss = trial.suggest_float('stop_loss', 0.005, 0.01)

    cerebro = bt.Cerebro()
    data = df.copy()
    data['datetime'] = pd.to_datetime(data['datetime'])
    data.set_index('datetime', inplace=True)
    cerebro.adddata(bt.feeds.PandasData(dataname=data))

    cerebro.addstrategy(MyStrategy, short_ma=short_ma, long_ma=long_ma,
                        take_profit=take_profit, stop_loss=stop_loss)
    cerebro.broker.set_cash(100000)
    cerebro.addsizer(bt.sizers.PercentSizer, percents=90)
    cerebro.broker.setcommission(commission=0.001)
    cerebro.run()
    return cerebro.broker.getvalue()


if __name__ == '__main__':
    # Список инструментов для оптимизации
    symbols = ['BTCUSDT', 'ETHUSDT', 'XRPUSDT', 'NG1!', 'SI1!',
               'EURUSD', 'USDJPY', 'GBPCAD', 'AUDUSD', 'SBER']

    results = []

    for symbol in symbols:
        try:
            print(f"Оптимизация для {symbol}...")
            df = get_tradingview_data(symbol, 2400, '1H')

            study = optuna.create_study(direction='maximize')
            # lambda передаёт df как аргумент в objective
            study.optimize(lambda trial: objective(trial, df), n_trials=5)

            trial = study.best_trial
            results.append({
                'symbol': symbol,
                'best_value': trial.value,
                'short_ma': trial.params['short_ma'],
                'long_ma': trial.params['long_ma'],
                'take_profit': trial.params['take_profit'],
                'stop_loss': trial.params['stop_loss']
            })
            print(f"Лучшие параметры для {symbol}: {trial.params}\n")

        except Exception as e:
            print(f"Ошибка при обработке {symbol}: {e}\n")

    # Сохранение результатов
    df_results = pd.DataFrame(results)
    df_results.to_csv('optuna_results.csv', index=False)
    print('Оптимизация завершена. Результаты сохранены в optuna_results.csv')
    print(df_results)
```

---

## 4. Важные замечания

- **`lambda trial: objective(trial, df)`** — единственный способ передать дополнительный аргумент в `objective`. Optuna передаёт только `trial`
- **`try / except`** — обязательно при работе с внешними источниками данных. Если тикер недоступен — цикл продолжается
- **`n_trials=5`** — очень мало для реальной оптимизации. Для production: 100-500 триалов на инструмент
- **Пример результата**: SBER → 128 576 (+28%), ETHUSDT → 117 213 (+17%), NG1! → 89 222 (-11%)

## 5. Шпаргалка

| Действие | Код |
|----------|-----|
| Передать df в objective | `study.optimize(lambda trial: objective(trial, df), n_trials=N)` |
| Отключить лог Optuna | `optuna.logging.set_verbosity(optuna.logging.WARNING)` |
| Сохранить результаты | `df_results.to_csv('optuna_results.csv', index=False)` |
| Защита от ошибок | `try: ... except Exception as e: print(e)` |
| Лучший результат | `study.best_trial.value`, `study.best_trial.params` |

## 6. Применение в боте (T-Invest API)

- **Портфельная оптимизация** — запускать этот скрипт для всех инструментов в портфеле. Каждый получает свои оптимальные параметры
- **Обновление параметров** — раз в месяц перезапускать оптимизацию на свежих данных через T-Invest API и обновлять конфиги бота
- **Параллельный запуск** — Optuna поддерживает многопоточность: `study.optimize(..., n_jobs=-1)` использует все ядра CPU
- **CSV как конфиг** — `optuna_results.csv` читается ботом при старте: каждый инструмент получает свои параметры из файла
