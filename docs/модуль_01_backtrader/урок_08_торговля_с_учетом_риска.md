# Урок 08: Торговля в Backtrader с учётом риска

## 1. Краткое резюме

Урок посвящён созданию кастомного сайзера для фьючерсов, который рассчитывает размер позиции с учётом риска на сделку (% от депозита), гарантийного обеспечения (ГО) и стоимости шага цены. Разбираются настройки `cerebro.broker.setcommission` для фьючерсов через параметры `margin` и `mult`. Данные загружаются через `price_loaders.tradingview` на примере нефти Brent (BR1!).

---

## 2. Ключевые концепции

- **Кастомный сайзер** — класс, наследующий `bt.Sizer` с методом `_getsizing`. Рассчитывает количество контрактов на основе риска и ГО
- **ГО (Гарантийное Обеспечение)** — сумма, блокируемая брокером на один фьючерсный контракт. Задаётся через `margin` в `setcommission`
- **`mult`** — стоимость шага цены / размер шага. Для нефти Brent: `mult=920` (стоимость пункта в рублях)
- **Риск на сделку** — `risk_percent` % от депозита. Размер позиции ограничивается и по риску, и по ГО
- **Режимы расчёта** — `sizing_mode`: `'min'` (консервативный), `'max'` (агрессивный), `'avg'` (средний)

---

## 3. Код урока

```python
import backtrader as bt
import pandas as pd
import datetime


class FuturesRiskSizer(bt.Sizer):
    """Сайзер для фьючерсов с учётом риска и ГО"""
    params = (
        ('risk_percent', 1.0),      # Риск на сделку (% от депозита)
        ('max_capital_use', 50.0),  # Максимальное использование депозита (%)
        ('margin', 16000),          # Гарантийное обеспечение на контракт
        ('mult', 920),              # Стоимость шага цены / шаг цены
        ('sizing_mode', 'min'),     # 'min', 'max', 'avg'
    )

    def log_sizing(self, txt):
        print(f"SIZER: {txt}")

    def _getsizing(self, comminfo, cash, data, isbuy):
        portfolio_value = self.broker.getvalue()
        self.log_sizing(f"Размер портфеля: {portfolio_value:.2f}")

        max_risk = portfolio_value * (self.p.risk_percent / 100.0)
        self.log_sizing(f"Максимальный риск: {max_risk:.2f}")

        current_price = data[0]

        if hasattr(self.strategy, 'stoploss'):
            stop_distance = current_price * self.strategy.params.stoploss
            risk_per_contract = stop_distance * self.p.mult
            size_by_risk = int(max_risk / risk_per_contract)
        else:
            size_by_risk = int(portfolio_value // self.p.margin)

        max_contracts_by_margin = int(
            (portfolio_value * (self.p.max_capital_use / 100.0)) / self.p.margin
        )

        if self.p.sizing_mode == 'min':
            final_size = min(size_by_risk, max_contracts_by_margin)
        elif self.p.sizing_mode == 'max':
            final_size = max(size_by_risk, max_contracts_by_margin)
        else:  # 'avg'
            final_size = (size_by_risk + max_contracts_by_margin) // 2

        self.log_sizing(f"Итоговый размер: {final_size} контрактов")
        return max(1, min(final_size, max_contracts_by_margin))


class Strategy3(bt.Strategy):
    params = (
        ('stoploss', 0.01),    # 1% стоп-лосс
        ('printlog', False),
        ('margin', 16000),
    )

    def log(self, txt, dt=None, doprint=False):
        if self.params.printlog or doprint:
            dt = dt or self.datas[0].datetime.datetime(0)
            print('%s, %s' % (dt.isoformat(), txt))

    def __init__(self):
        self.dataclose = self.datas[0].close
        self.order = None
        self.buyprice = None
        self.stop_price = None
        self.rsi = bt.indicators.RelativeStrengthIndex()

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [order.Completed]:
            if order.isbuy():
                self.log('Покупка: Цена: %.2f, Кол-во: %.2f, Сумма: %.2f, Комиссия: %.2f' %
                         (order.executed.price, order.executed.size,
                          order.executed.value, order.executed.comm))
                self.buyprice = order.executed.price
                self.stop_price = order.executed.price * (1 - self.params.stoploss)
            else:
                self.log('Продажа: Цена: %.2f, Кол-во: %.2f, Сумма: %.2f, Комиссия: %.2f' %
                         (order.executed.price, order.executed.size,
                          order.executed.value, order.executed.comm))
                self.stop_price = order.executed.price * (1 + self.params.stoploss)
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.log('Отмена ордера')
        self.order = None

    def notify_trade(self, trade):
        if not trade.isclosed:
            return
        self.log('P&L: Общая %.2f, С комиссией %.2f' % (trade.pnl, trade.pnlcomm))

    def next(self):
        if self.order:
            return

        if not self.position:
            if self.rsi[0] >= 30 and self.rsi[-1] < 30:
                self.log('Покупка: %.2f' % self.dataclose[0])
                self.order = self.buy()

            elif self.rsi[-1] > 70 and self.rsi[0] <= 70:
                self.log('Продажа: %.2f' % self.dataclose[0])
                self.order = self.sell()

        else:
            if self.position.size > 0:
                if (self.rsi[0] >= 70 or
                        self.position.price / self.dataclose[0] > (1 + self.params.stoploss)):
                    self.order = self.sell(size=abs(self.position.size))
            else:
                if (self.rsi[0] <= 30 or
                        self.position.price / self.dataclose[0] < (1 - self.params.stoploss)):
                    self.order = self.buy(size=abs(self.position.size))


if __name__ == '__main__':
    # Загрузка данных BR1! через price_loaders.tradingview
    # data_br = load_asset_price("BR1!", 6000, "1H", None)
    # data_br = data_br.rename(columns={'time': 'datetime'})

    cerebro = bt.Cerebro()
    cerebro.addstrategy(Strategy3, printlog=True)
    cerebro.addanalyzer(bt.analyzers.PyFolio, _name='PyFolio')
    cerebro.addanalyzer(bt.analyzers.TradeAnalyzer, _name='TradeAnalyzer')

    # data = data_br.copy()
    # data["datetime"] = pd.to_datetime(data["datetime"]) + datetime.timedelta(hours=7)
    # data.set_index("datetime", inplace=True)
    # cerebro.adddata(bt.feeds.PandasData(dataname=data))

    cerebro.broker.setcash(100000.0)
    # Настройка для фьючерса: margin=ГО, mult=стоимость шага цены/шаг цены
    cerebro.broker.setcommission(commission=0.001, margin=16000, mult=920)

    cerebro.addsizer(
        FuturesRiskSizer,
        risk_percent=1.0,        # Риск 1% на сделку
        max_capital_use=50.0,    # Максимум 50% депозита
        margin=16000,
        mult=920
    )

    print('Стартовый портфель: %.2f' % cerebro.broker.getvalue())
    results = cerebro.run()
    print('Конечный портфель: %.2f' % cerebro.broker.getvalue())
```

---

## 4. Важные замечания

- **`margin` и `mult` в `setcommission`** — обязательны для фьючерсов. Без них Backtrader не сможет корректно рассчитать P&L и маржу
- **`mult=920`** — для нефти Brent (BR): стоимость 1 пункта × размер шага. Для других фьючерсов своё значение
- **`_getsizing`** — вызывается перед каждым ордером. Получает `cash`, `data`, `isbuy` — возвращает количество контрактов
- **`hasattr(self.strategy, 'stoploss')`** — сайзер может обращаться к параметрам стратегии для расчёта риска на контракт

## 5. Шпаргалка

| Действие | Код |
|----------|-----|
| Настройка фьючерса | `cerebro.broker.setcommission(commission=0.001, margin=16000, mult=920)` |
| Кастомный сайзер | `class MySizer(bt.Sizer): def _getsizing(self, comminfo, cash, data, isbuy):` |
| Подключение сайзера | `cerebro.addsizer(FuturesRiskSizer, risk_percent=1.0, margin=16000)` |
| Размер портфеля в сайзере | `self.broker.getvalue()` |
| Контрактов по марже | `int(portfolio_value // margin)` |

## 6. Применение в боте (T-Invest API)

- **Расчёт размера позиции** — логику `FuturesRiskSizer` переносим в функцию расчёта лота перед отправкой ордера в T-Invest. Передаём текущий баланс счёта и параметры инструмента
- **ГО из API** — T-Invest API возвращает размер ГО для каждого фьючерса через `get_futures_margin`. Динамически подставляем в формулу
- **Контроль маржи** — перед открытием позиции проверяем `free_margin >= margin * size` чтобы брокер не отклонил ордер
