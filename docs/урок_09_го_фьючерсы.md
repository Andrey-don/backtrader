# Урок 09: ГО и торговля фьючерсами в Backtrader

## 1. Краткое резюме

Урок посвящён особенностям торговли фьючерсами в Backtrader: настройке гарантийного обеспечения (ГО), расчёту размера позиции от свободных средств и параметру `mult`. Рассматривается стратегия на основе RSI с ручным расчётом количества контрактов прямо в методе `next`. Данные — нефть Brent (BR1!) через `price_loaders.tradingview`.

---

## 2. Ключевые концепции

- **ГО (Гарантийное Обеспечение)** — сумма, блокируемая на каждый фьючерсный контракт. Задаётся в `setcommission(margin=N)`
- **Расчёт лота от ГО** — `size = self.broker.getvalue() // self.params.margin` — максимальное количество контрактов, которое можно открыть
- **`mult`** — коэффициент пересчёта пунктов в деньги. Для BR1!: `mult=920`
- **RSI как триггер** — вход при пересечении уровней 30 (покупка) и 70 (продажа)
- **Явное закрытие по размеру** — `self.sell(size=self.position.size)` вместо `self.close()` — для полного закрытия фьючерсной позиции

---

## 3. Код урока

```python
import backtrader as bt
import pandas as pd
import datetime


class Strategy3(bt.Strategy):
    params = (
        ('stoploss', 0.01),
        ('printlog', False),
        ('margin', 16000),  # ГО на контракт
    )

    def log(self, txt, dt=None, doprint=False):
        if self.params.printlog or doprint:
            dt = dt or self.datas[0].datetime.datetime(0)
            print('%s, %s' % (dt.isoformat(), txt))

    def __init__(self):
        self.dataclose = self.datas[0].close
        self.order = None
        self.buyprice = None
        self.rsi = bt.indicators.RelativeStrengthIndex()

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return
        if order.status in [order.Completed]:
            if order.isbuy():
                self.log('Покупка: Цена: %.2f, Кол-во: %.2f, Сумма: %.2f, Комиссия: %.2f' %
                         (order.executed.price, order.executed.size,
                          order.executed.value, order.executed.comm))
                self.buyprice = order.executed.price
            else:
                self.log('Продажа: Цена: %.2f, Кол-во: %.2f, Сумма: %.2f, Комиссия: %.2f' %
                         (order.executed.price, order.executed.size,
                          order.executed.value, order.executed.comm))
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.log('Отмена ордера')
        self.order = None

    def notify_trade(self, trade):
        if not trade.isclosed:
            return
        self.log('P&L: Общая %.2f, С комиссией %.2f' % (trade.pnl, trade.pnlcomm))

    def next(self):
        if not self.position:
            if self.rsi[0] >= 30 and self.rsi[-1] < 30:
                self.log('Покупка: %.2f' % self.dataclose[0])
                if self.order:
                    self.cancel(self.order)
                # Расчёт количества контрактов от ГО
                size_buy = self.broker.getvalue() // self.params.margin
                self.order = self.buy(size=size_buy)

            if self.rsi[-1] > 70 and self.rsi[0] <= 70:
                self.log('Продажа: %.2f' % self.dataclose[0])
                if self.order:
                    self.cancel(self.order)
                size_sell = self.broker.getvalue() // self.params.margin
                self.order = self.sell(size=size_sell)

        else:
            if self.position.size > 0 and self.rsi[0] >= 70:
                self.log('Закрытие лонга: %.2f' % self.dataclose[0])
                self.order = self.sell(size=self.position.size)

            elif self.position.size > 0 and \
                    self.position.price / self.dataclose[0] > (1 + self.params.stoploss):
                self.log('Стоп лонга: %.2f' % self.dataclose[0])
                self.order = self.sell(size=self.position.size)

            elif self.position.size < 0 and self.rsi[0] <= 30:
                self.log('Закрытие шорта: %.2f' % self.dataclose[0])
                self.order = self.buy(size=abs(self.position.size))

            elif self.position.size < 0 and \
                    self.position.price / self.dataclose[0] < (1 - self.params.stoploss):
                self.log('Стоп шорта: %.2f' % self.dataclose[0])
                self.order = self.buy(size=abs(self.position.size))


if __name__ == '__main__':
    # data_br = load_asset_price("BR1!", 6000, "1H", None)
    # data_br = data_br.rename(columns={'time': 'datetime'})

    cerebro = bt.Cerebro()
    cerebro.addstrategy(Strategy3, printlog=True)
    cerebro.addanalyzer(bt.analyzers.PyFolio, _name='PyFolio')
    cerebro.addanalyzer(bt.analyzers.TradeAnalyzer, _name='TradeAnalyzer')

    # data = data_br.copy()
    # data["datetime"] = pd.to_datetime(data["datetime"]) + datetime.timedelta(hours=7)
    # data.set_index("datetime", inplace=True)
    # cerebro.adddata(bt.feeds.PandasData(dataname=data))

    cerebro.broker.setcash(100000.0)
    # margin=ГО, mult=стоимость шага цены / шаг цены
    cerebro.broker.setcommission(commission=0.001, margin=16000, mult=920)

    print('Стартовый портфель: %.2f' % cerebro.broker.getvalue())
    results = cerebro.run()
    print('Конечный портфель: %.2f' % cerebro.broker.getvalue())
```

---

## 4. Важные замечания

- **Отличие от урока 08** — здесь размер лота считается прямо в `next` через `self.broker.getvalue() // margin`, без кастомного сайзера
- **`self.cancel(self.order)`** — перед новым ордером отменяем старый, если он ещё в очереди
- **Стоп через соотношение цен** — `self.position.price / self.dataclose[0] > (1 + stop)` — нестандартная проверка (обычно сравнивают `close < position.price * (1 - stop)`)

---

## 5. Шпаргалка

| Действие | Код |
|----------|-----|
| Настройка фьючерса | `cerebro.broker.setcommission(commission=0.001, margin=16000, mult=920)` |
| Лот от ГО | `size = self.broker.getvalue() // self.params.margin` |
| Закрытие по размеру | `self.sell(size=self.position.size)` |
| Отмена ордера | `self.cancel(self.order)` |
| Текущий баланс | `self.broker.getvalue()` |

---

## 6. Применение в боте (T-Invest API)

- **Расчёт лота** — перед каждым ордером запрашиваем баланс счёта через T-Invest API и делим на ГО инструмента
- **ГО динамически** — ГО фьючерсов меняется (особенно в волатильные дни). Каждый раз брать актуальное значение через `InstrumentsService`
- **Явный размер в ордере** — T-Invest требует указывать `quantity` явно. Передаём рассчитанный `size`
